def classes(path,names,ctx):

fan_tests=module('audit_fan_tests',GTS/'tests/test_fan.py'); protocol=fan_tests.protocol

async def late_failure():
    fan,c=fan_tests.FanTests().make_fan()
    fan._run_mode_before_off=protocol.MANUAL_OR_AUTO_TIDE
    c.data.current_mode=protocol.MODE_OFF
    calls=[]
    async def write(name,value,sw):
        calls.append((name,value))
        if len(calls)==4: raise OSError('last tide write lost')
    c.device.async_write_named=write
    try: await fan.async_turn_on()
    except OSError: pass
    assert calls[:3]==[('manual_or_auto',1),('manual_mode',1),('manual_power_1',80)]
    print('GTS late failure: manual mode and power 80 applied before final tide write fails; no recovery write follows',calls)

async def interleave():
    fan,c=fan_tests.FanTests().make_fan()
    fan._run_mode_before_off=protocol.MANUAL_OR_AUTO_TIDE
    writes=[]; first=asyncio.Event();release=asyncio.Event()
    async def write(name,value,sw):
        writes.append((name,value))
        if len(writes)==1: first.set();await release.wait()
    c.device.async_write_named=write
    turning=asyncio.create_task(fan.async_turn_on()); await first.wait()
    await fan.async_turn_off();release.set();await turning
    stop=writes.index(('manual_mode',7))
    assert any(name=='manual_mode' and value in protocol.MANUAL_MODES for name,value in writes[stop+1:])
    print('GTS interleave: a later turn_off is overwritten by unfinished earlier turn_on',writes)
